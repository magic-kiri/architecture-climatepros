# Step 2 — Network
# Run these in PowerShell after loading vars.ps1

# 1. Create VPC
$env:VPC_ID = (aws ec2 create-vpc --cidr-block 10.42.0.0/16 `
  --tag-specifications 'ResourceType=vpc,Tags=[{Key=Name,Value=cp-test-vpc},{Key=Project,Value=stream1},{Key=Env,Value=test},{Key=ManagedBy,Value=techjays}]' `
  --query 'Vpc.VpcId' --output text --profile cp-test)
echo $env:VPC_ID

# 2. Enable DNS (required for ElastiCache endpoint resolution)
aws ec2 modify-vpc-attribute --vpc-id $env:VPC_ID --enable-dns-hostnames --profile cp-test

# 3. Create two subnets in different AZs
$env:SUBNET_A = (aws ec2 create-subnet --vpc-id $env:VPC_ID --cidr-block 10.42.1.0/24 `
  --availability-zone ($env:AWS_REGION + "a") `
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=cp-test-a},{Key=Project,Value=stream1},{Key=Env,Value=test}]' `
  --query 'Subnet.SubnetId' --output text --profile cp-test)
echo $env:SUBNET_A

$env:SUBNET_B = (aws ec2 create-subnet --vpc-id $env:VPC_ID --cidr-block 10.42.2.0/24 `
  --availability-zone ($env:AWS_REGION + "b") `
  --tag-specifications 'ResourceType=subnet,Tags=[{Key=Name,Value=cp-test-b},{Key=Project,Value=stream1},{Key=Env,Value=test}]' `
  --query 'Subnet.SubnetId' --output text --profile cp-test)
echo $env:SUBNET_B

# 4. Internet gateway
$env:IGW_ID = (aws ec2 create-internet-gateway `
  --tag-specifications 'ResourceType=internet-gateway,Tags=[{Key=Name,Value=cp-test-igw},{Key=Project,Value=stream1},{Key=Env,Value=test}]' `
  --query 'InternetGateway.InternetGatewayId' --output text --profile cp-test)
echo $env:IGW_ID

aws ec2 attach-internet-gateway --vpc-id $env:VPC_ID --internet-gateway-id $env:IGW_ID --profile cp-test

# 5. Route table — send all outbound traffic to the gateway
$env:RTB_ID = (aws ec2 create-route-table --vpc-id $env:VPC_ID `
  --tag-specifications 'ResourceType=route-table,Tags=[{Key=Name,Value=cp-test-rtb},{Key=Project,Value=stream1},{Key=Env,Value=test}]' `
  --query 'RouteTable.RouteTableId' --output text --profile cp-test)

aws ec2 create-route --route-table-id $env:RTB_ID --destination-cidr-block 0.0.0.0/0 --gateway-id $env:IGW_ID --profile cp-test
aws ec2 associate-route-table --route-table-id $env:RTB_ID --subnet-id $env:SUBNET_A --profile cp-test
aws ec2 associate-route-table --route-table-id $env:RTB_ID --subnet-id $env:SUBNET_B --profile cp-test

# 6. Print all IDs to copy into vars.ps1
echo "VPC_ID:    $env:VPC_ID"
echo "SUBNET_A:  $env:SUBNET_A"
echo "SUBNET_B:  $env:SUBNET_B"
echo "IGW_ID:    $env:IGW_ID"
echo "RTB_ID:    $env:RTB_ID"

# =========================================================
# Step 3 — Security Groups
# =========================================================

# 1. Worker SG (Fargate tasks)
$env:WORKER_SG_ID = (aws ec2 create-security-group `
  --group-name cp-test-workers --description "Dispatch Fargate workers" --vpc-id $env:VPC_ID `
  --tag-specifications 'ResourceType=security-group,Tags=[{Key=Project,Value=stream1},{Key=Env,Value=test},{Key=ManagedBy,Value=techjays}]' `
  --query 'GroupId' --output text --profile cp-test)
echo $env:WORKER_SG_ID

# 2. Redis SG
$env:REDIS_SG_ID = (aws ec2 create-security-group `
  --group-name cp-test-redis --description "Dispatch queue - ElastiCache" --vpc-id $env:VPC_ID `
  --tag-specifications 'ResourceType=security-group,Tags=[{Key=Project,Value=stream1},{Key=Env,Value=test},{Key=ManagedBy,Value=techjays}]' `
  --query 'GroupId' --output text --profile cp-test)
echo $env:REDIS_SG_ID

# 3. Allow workers to reach Redis on 6379
aws ec2 authorize-security-group-ingress --group-id $env:REDIS_SG_ID `
  --protocol tcp --port 6379 --source-group $env:WORKER_SG_ID --profile cp-test

# 4. Allow all inbound traffic to workers from anywhere (test — for public debugging)
aws ec2 authorize-security-group-ingress --group-id $env:WORKER_SG_ID `
  --protocol -1 --cidr 0.0.0.0/0 --profile cp-test

# 5. Print IDs to copy into vars.ps1
echo "WORKER_SG_ID: $env:WORKER_SG_ID"
echo "REDIS_SG_ID:  $env:REDIS_SG_ID"

# =========================================================
# Step 4 — Secret (Redis Auth Token)
# =========================================================

# 1. Generate a random 64-char hex token (PowerShell — no openssl needed)
$env:REDIS_AUTH_TOKEN = -join ((1..32) | ForEach-Object { '{0:x2}' -f (Get-Random -Maximum 256) })

# 2. Store it in Secrets Manager
aws secretsmanager create-secret --name cp/test/redis-auth-token `
  --secret-string $env:REDIS_AUTH_TOKEN `
  --tags Key=Project,Value=stream1 Key=Env,Value=test Key=ManagedBy,Value=techjays `
  --query 'ARN' --output text --profile cp-test

# NOTE: Do NOT close this terminal — REDIS_AUTH_TOKEN is needed in Step 6.
# For safety, also save it somewhere private (e.g. your password manager).
# Do NOT put it in vars.ps1 if that file is anywhere near git.

# =========================================================
# Step 5 — IAM Roles
# Run from: E:\Code\climatepros\architecture-climatepros\infra-setup\
# =========================================================

# 1. Write trust policy file (no BOM — Out-File utf8 adds BOM, WriteAllText does not)
$t = @'
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Principal": { "Service": "ecs-tasks.amazonaws.com" },
    "Action": "sts:AssumeRole"
  }]
}
'@
[System.IO.File]::WriteAllText("$PWD\trust-ecs.json", $t)

# 2. Write execution role inline policy (read secrets)
$e = @"
{
  "Version": "2012-10-17",
  "Statement": [{
    "Effect": "Allow",
    "Action": "secretsmanager:GetSecretValue",
    "Resource": "arn:aws:secretsmanager:$($env:AWS_REGION):$($env:ACCOUNT_ID):secret:cp/$($env:ENV)/*"
  }]
}
"@
[System.IO.File]::WriteAllText("$PWD\exec-role-policy.json", $e)

# 3. Write task role inline policy (read secrets + write logs/metrics)
$k = @"
{
  "Version": "2012-10-17",
  "Statement": [
    {
      "Effect": "Allow",
      "Action": "secretsmanager:GetSecretValue",
      "Resource": "arn:aws:secretsmanager:$($env:AWS_REGION):$($env:ACCOUNT_ID):secret:cp/$($env:ENV)/*"
    },
    {
      "Effect": "Allow",
      "Action": ["logs:PutLogEvents","logs:CreateLogStream","cloudwatch:PutMetricData"],
      "Resource": "*"
    }
  ]
}
"@
[System.IO.File]::WriteAllText("$PWD\task-role-policy.json", $k)

# 4. Create execution role
aws iam create-role --role-name cp-test-task-exec `
  --assume-role-policy-document file://trust-ecs.json `
  --tags Key=Project,Value=stream1 Key=Env,Value=test Key=ManagedBy,Value=techjays `
  --profile cp-test

# Attach AWS managed policy (image pull + logs)
aws iam attach-role-policy --role-name cp-test-task-exec `
  --policy-arn arn:aws:iam::aws:policy/service-role/AmazonECSTaskExecutionRolePolicy `
  --profile cp-test

# Attach inline policy (read secrets)
aws iam put-role-policy --role-name cp-test-task-exec --policy-name read-secrets `
  --policy-document file://exec-role-policy.json `
  --profile cp-test

# 5. Create task role
aws iam create-role --role-name cp-test-task `
  --assume-role-policy-document file://trust-ecs.json `
  --tags Key=Project,Value=stream1 Key=Env,Value=test Key=ManagedBy,Value=techjays `
  --profile cp-test

# Attach inline policy (secrets + logs/metrics)
aws iam put-role-policy --role-name cp-test-task --policy-name worker `
  --policy-document file://task-role-policy.json `
  --profile cp-test

echo "Done. If you get AccessDenied on any step, wait 5s and retry once."

# =========================================================
# Step 6 — ElastiCache Redis
# =========================================================

# 0. Confirm auth token is still set
echo "REDIS_AUTH_TOKEN: $env:REDIS_AUTH_TOKEN"
# If blank — retrieve it from Secrets Manager:
# $env:REDIS_AUTH_TOKEN = (aws secretsmanager get-secret-value --secret-id cp/test/redis-auth-token --query SecretString --output text --profile cp-test)

# 1. Subnet group (same two subnets as the tasks)
aws elasticache create-cache-subnet-group `
  --cache-subnet-group-name cp-test-redis-subnets `
  --cache-subnet-group-description "Dispatch queue subnets" `
  --subnet-ids $env:SUBNET_A $env:SUBNET_B `
  --profile cp-test

# 2. Parameter group — noeviction so the queue never silently drops jobs
aws elasticache create-cache-parameter-group `
  --cache-parameter-group-name cp-test-redis7 `
  --cache-parameter-group-family redis7 `
  --description "noeviction - dispatch queue must not lose entries" `
  --profile cp-test

aws elasticache modify-cache-parameter-group `
  --cache-parameter-group-name cp-test-redis7 `
  --parameter-name-values "ParameterName=maxmemory-policy,ParameterValue=noeviction" `
  --profile cp-test

# 3. Create Redis cluster (takes 5-10 min)
aws elasticache create-replication-group `
  --replication-group-id cp-test-dispatch-redis `
  --replication-group-description "Auto-Dispatch queue (Redis Streams)" `
  --engine redis --engine-version 7.1 `
  --cache-node-type cache.t4g.small `
  --num-cache-clusters 1 `
  --cache-subnet-group-name cp-test-redis-subnets `
  --cache-parameter-group-name cp-test-redis7 `
  --security-group-ids $env:REDIS_SG_ID `
  --transit-encryption-enabled --transit-encryption-mode required `
  --at-rest-encryption-enabled `
  --auth-token "$env:REDIS_AUTH_TOKEN" `
  --snapshot-retention-limit 7 `
  --tags Key=Project,Value=stream1 Key=Env,Value=test Key=ManagedBy,Value=techjays `
  --profile cp-test

# 4. Wait for Redis to be ready (blocks until available)
aws elasticache wait replication-group-available `
  --replication-group-id cp-test-dispatch-redis `
  --profile cp-test

# 5. Capture endpoint — add to vars.ps1
$env:REDIS_ENDPOINT = (aws elasticache describe-replication-groups `
  --replication-group-id cp-test-dispatch-redis `
  --query 'ReplicationGroups[0].NodeGroups[0].PrimaryEndpoint.Address' `
  --output text --profile cp-test)
echo "REDIS_ENDPOINT: $env:REDIS_ENDPOINT"

# =========================================================
# Step 7 — ECR + ECS
# =========================================================

# 1. Create ECR repository
aws ecr create-repository --repository-name cp/dispatch-worker `
  --image-scanning-configuration scanOnPush=true `
  --image-tag-mutability IMMUTABLE `
  --tags Key=Project,Value=stream1 Key=ManagedBy,Value=techjays `
  --profile cp-test

# Keep only 10 most recent images
[System.IO.File]::WriteAllText("$PWD\lifecycle.json", '{"rules":[{"rulePriority":1,"description":"keep 10","selection":{"tagStatus":"any","countType":"imageCountMoreThan","countNumber":10},"action":{"type":"expire"}}]}')
aws ecr put-lifecycle-policy --repository-name cp/dispatch-worker `
  --lifecycle-policy-text file://lifecycle.json `
  --region $env:AWS_REGION --profile cp-test

# 2. First image push — Docker Desktop must be running; cd into your worker code directory first
$env:ECR_URI = "$env:ACCOUNT_ID.dkr.ecr.$env:AWS_REGION.amazonaws.com/cp/dispatch-worker"

aws ecr get-login-password --region $env:AWS_REGION --profile cp-test `
  | docker login --username AWS --password-stdin "$env:ACCOUNT_ID.dkr.ecr.$env:AWS_REGION.amazonaws.com"

docker build --platform linux/amd64 -t "${env:ECR_URI}:test-0.1.0" .
docker push "${env:ECR_URI}:test-0.1.0"

# 3. Log group (must exist before cluster or task fails to start)
aws logs create-log-group --log-group-name /ecs/cp-test-dispatch-worker --profile cp-test
aws logs put-retention-policy --log-group-name /ecs/cp-test-dispatch-worker `
  --retention-in-days 30 --profile cp-test

# 4. ECS Cluster
aws ecs create-cluster --cluster-name cp-test-dispatch `
  --settings name=containerInsights,value=enabled `
  --tags key=Project,value=stream1 key=Env,value=test key=ManagedBy,value=techjays `
  --profile cp-test

# 5. Get secret ARN (needed in task definition)
$env:SECRET_ARN = (aws secretsmanager describe-secret `
  --secret-id cp/test/redis-auth-token `
  --query ARN --output text --profile cp-test)
echo "SECRET_ARN: $env:SECRET_ARN"

# 6. Write task definition JSON
$taskdef = @"
